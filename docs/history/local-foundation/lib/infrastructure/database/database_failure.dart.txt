import 'dart:io';
import 'package:drift/isolate.dart';
import 'package:sqlite3/sqlite3.dart';
import '../../core/foundation_failure.dart';

FoundationFailure safeDatabaseFailure(Object error) {
  if (error is DriftRemoteException) {
    return safeDatabaseFailure(error.remoteCause);
  }
  if (error is FoundationFailure) return FoundationFailure(error.kind);
  if (error is FileSystemException) {
    return const FoundationFailure(FoundationFailureKind.storageFailure);
  }
  if (error is SqliteException) {
    return FoundationFailure(switch (error.resultCode) {
      // NOTADB cannot distinguish a wrong key from a damaged first page.
      26 => FoundationFailureKind.invalidKey,
      11 => FoundationFailureKind.integrityFailure,
      8 || 10 || 13 || 14 => FoundationFailureKind.storageFailure,
      _ => FoundationFailureKind.databaseUnavailable,
    });
  }
  return const FoundationFailure(FoundationFailureKind.databaseUnavailable);
}
